S = load_S("inc_star_deg")
A = load_A()
N = 2000
SG = np.arange(0.0, 4.0 + 1e-9, 0.025)
i1 = int(np.argmin(abs(SG - 1.0)))
n = len(S.R)
xK = np.clip(S.R / S.Reff - 1, 0, 4)
aK = -0.146 * xK ** 2 + 1.204 * xK + 1.475
Ez = M.E_of_z(S.z)
A0c = M.A0["canonical"]
Rm = S.R * M.KPC
G = M.G; MS = M.MSUN
nu = M.nu_mono


def enc(y):
    return 1.0 - np.exp(-y) * (1.0 + y)


def gbar_v(logM, mu, gas_scale=2.0):
    Ms = 10.0 ** logM
    Mb = Ms * enc(S.R / S.Rd) + mu * Ms * enc(S.R / (gas_scale * S.Rd))
    return G * Mb * MS / Rm ** 2


# analysis-side fixed pieces (nominal mass, mu = 0.67)
gb_a = gbar_v(S.logM, 0.67)
a_flat = A0c * np.ones(n); a_H = A0c * Ez
gp_f = nu(gb_a / a_flat) * gb_a
gp_h = nu(gb_a / a_H) * gb_a
em_f = S.mass_err * np.abs(M.slope(gb_a, a_flat))
em_h = S.mass_err * np.abs(M.slope(gb_a, a_H))
GR = grid_arrays(S, A, SG)          # real anchor offset per s (and per-galaxy real KURVS arrays for the validation)
Af, Ae = GR["Af"], GR["Ae"]


def analyse(V, sig):
    """V, sig: (N, n) observed. returns Delta'_flat, sigma_flat, Delta'_H, sigma_H each (N, ns)"""
    s = SG[None, :, None]
    Vc2 = V[:, None, :] ** 2 + s * aK[None, None, :] * sig[:, None, :] ** 2
    gobs = Vc2 * 1e6 / Rm[None, None, :]
    ev = (2 * V[:, None, :] / Vc2 * S.eV[None, None, :]) / LN10
    dC = 2 * s * aK[None, None, :] * sig[:, None, :]
    es = np.abs(dC) * S.esig[None, None, :] / Vc2 / LN10
    ei = (2.0 * (V[:, None, :] ** 2 / Vc2) / LN10) * (1.0 / np.tan(S.inc))[None, None, :] * S.einc[None, None, :]
    out = []
    for gp, em in ((gp_f, em_f), (gp_h, em_h)):
        d = np.log10(gobs) - np.log10(gp)[None, None, :]
        e = np.sqrt(ev ** 2 + es ** 2 + ei ** 2 + em[None, None, :] ** 2)
        m, err = pool_v(d, e)
        out.append((m - Af[None, :], np.sqrt(err ** 2 + Ae[None, :] ** 2)))
    return out[0][0], out[0][1], out[1][0], out[1][1]


# --- validation on the real data

P(f"CFG168 null; repo = <repo>\nC-d0 vectorised analysis vs shared module on the real data (s = 0, 1, 2.5, 4): max diff {dd:.2e} -> {'PASS' if dd < 1e-9 else 'FAIL'}")

def smid_of(df, dh):
    """per mock: (s_mid or nan, code) ; code 0 crossing found, -1 below range (sum>0 at s=0), +1 above range (sum<0 at s=4)"""
    y = df + dh
    Nn = y.shape[0]
    out = np.full(Nn, np.nan); code = np.zeros(Nn, int)
    for k in range(Nn):
        yy = y[k]
        if yy[0] > 0:
            code[k] = -1; continue
        j = np.where(yy > 0)[0]
        if len(j) == 0:
            code[k] = 1; continue
        j = j[0]
        out[k] = SG[j - 1] + (0 - yy[j - 1]) * (SG[j] - SG[j - 1]) / (yy[j] - yy[j - 1])
    return out, code


def generate(law, rng, mu_med, mu_sd_dex, s_true, s_pg, s_coh, N=N):
    """returns observed V, sig (N,n) and the floored fraction"""
    dm = rng.normal(0, S.mass_err, (N, n))                          # true mass offset (dex)
    if mu_sd_dex > 0:
        mu = mu_med * 10 ** rng.normal(0, mu_sd_dex, (N, n))
    else:
        mu = mu_med * np.ones((N, n))
    Ms = 10.0 ** (S.logM + dm)
    Mb = Ms * enc(S.R / S.Rd) + mu * Ms * enc(S.R / (2.0 * S.Rd))
    gb = G * Mb * M.MSUN / Rm ** 2
    a = A0c * (np.ones(n) if law == "flat" else Ez)
    g = nu(gb / a) * gb
    Vc2 = g * Rm / 1e6
    fac = s_true * np.exp(rng.normal(0, s_pg, (N, n)) + rng.normal(0, s_coh, (N, 1))) if (s_pg > 0 or s_coh > 0) else s_true * np.ones((N, n))
    sig_t = S.sig[None, :] * np.ones((N, 1))
    C = fac * aK[None, :] * sig_t ** 2
    V2 = Vc2 - C
    floor = 0.0025 * Vc2
    fl = V2 < floor
    V2 = np.where(fl, floor, V2)
    V = np.sqrt(V2)
    dinc = rng.normal(0, 1, (N, n)) * S.einc[None, :]
    Vobs = V * np.sin(S.inc)[None, :] / np.sin(S.inc[None, :] + dinc) + rng.normal(0, 1, (N, n)) * S.eV[None, :]
    sigo = np.maximum(sig_t + rng.normal(0, 1, (N, n)) * S.esig[None, :], 1.0)
    return Vobs, sigo, float(fl.mean())


FAM = {
    "ideal": dict(mu_med=0.67, mu_sd_dex=0.0, s_true=1.0, s_pg=0.0, s_coh=0.0),
    "N1": dict(mu_med=1.0, mu_sd_dex=0.3, s_true=1.0, s_pg=0.4, s_coh=0.2),
