-- T1_deep_drift: log c instead of log c / 2
theorem m06 {f a0 c g : ℝ} (hf : 0 < f) (ha : 0 < a0) (hc : 0 < c) (hg : 0 < g) : gObs nuDeep (c * f) a0 g = Real.sqrt c * gObs nuDeep f a0 g ∧ gObs nuDeep f (c * a0) g = Real.sqrt c * gObs nuDeep f a0 g ∧ Real.log (gObs nuDeep (c * f) a0 g) = Real.log (gObs nuDeep f a0 g) + Real.log c ∧ Real.log (gObs nuDeep f (c * a0) g) = Real.log (gObs nuDeep f a0 g) + Real.log c := T1_deep_drift hf ha hc hg
