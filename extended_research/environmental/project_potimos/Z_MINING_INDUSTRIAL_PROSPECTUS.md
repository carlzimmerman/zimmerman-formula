### The Selectivity Mechanism

Standard membranes cannot separate Li⁺ from Na⁺ because their **hydrated** radii are similar:
- Li⁺ (hydrated): 3.82 Å
- Na⁺ (hydrated): 3.58 Å

Z-Mining achieves selectivity through **partial desolvation**:

```
Li⁺ (hydrated)  →  [Z/2 pore]  →  Li⁺ (partial desolvation)  →  CAPTURE
     3.82 Å          2.89 Å              ~2.3 Å

Na⁺ (hydrated)  →  [Z/2 pore]  →  BLOCKED (ionic radius 1.02 Å too large)
     3.58 Å          2.89 Å
```

The Berry curvature at the pore edge creates a **Topological Potential Well**:
- Energetically favorable for Li⁺ (small bare ion)
- High repulsion barrier for Na⁺ (larger bare ion)

## Appendix: Z-Mining Algorithm

```python
#!/usr/bin/env python3
"""
Z-Mining Optimization Algorithm
Project Potimos v11.4.0

License: AGPL-3.0 | Patent Pending
Author: Carl Zimmerman
"""

import numpy as np

# Universal Constants
Z = np.sqrt(32 * np.pi / 3)  # 5.7888 Å
Z_HALF = Z / 2               # 2.8944 Å (Li⁺ capture pore)

# Ion Properties
IONS = {
    'Li+': {'ionic_r': 0.76, 'hydrated_r': 3.82, 'value_usd_kg': 70},
    'Na+': {'ionic_r': 1.02, 'hydrated_r': 3.58, 'value_usd_kg': 0.5},
    'Nd3+': {'ionic_r': 0.98, 'hydrated_r': 4.52, 'value_usd_kg': 150},
    'Co2+': {'ionic_r': 0.74, 'hydrated_r': 4.23, 'value_usd_kg': 30},
}

# System Parameters
CHERN_NUMBER = 0.96      # Topological invariant
ALIVENESS = 0.018        # Anti-fouling offset (1.8%)
BASE_EFFICIENCY = 0.92   # Baseline recovery

def calculate_capture_efficiency(target_ion: str, pore_diameter: float) -> float:
    """
    Calculate topological capture cross-section for target ion.

    The Berry curvature creates a potential well whose depth depends on
    the geometric match between ion radius and pore diameter.
    """
    ion = IONS.get(target_ion, IONS['Li+'])
    ionic_r = ion['ionic_r']

    # Geometric matching factor
    # Optimal when pore ~ 2 × ionic radius (allows desolvation)
    optimal_pore = 2 * ionic_r + 1.0  # +1Å for partial hydration shell
    geometric_match = 1.0 - abs(pore_diameter - optimal_pore) / optimal_pore
    geometric_match = max(0, geometric_match)

    # Topological enhancement from Chern number
    topo_factor = CHERN_NUMBER * (1 + ALIVENESS)

    # Combined capture efficiency
    efficiency = geometric_match * topo_factor * BASE_EFFICIENCY

    return min(0.99, efficiency)

def calculate_selectivity(target_ion: str, competitor_ion: str,
                          pore_diameter: float) -> float:
    """
    Calculate selectivity ratio between target and competitor ion.
    """
    eff_target = calculate_capture_efficiency(target_ion, pore_diameter)
    eff_competitor = calculate_capture_efficiency(competitor_ion, pore_diameter)

    if eff_competitor < 0.001:
        return 1000  # Effectively infinite selectivity

    return eff_target / eff_competitor

def project_annual_revenue(
    ion_concentration_ppm: float,
    flow_rate_lpm: float = 15.0,
    target_ion: str = 'Li+',
    pore_diameter: float = Z_HALF
) -> dict:
    """
    Project annual revenue from Z-Mining module.

    Parameters:
    -----------
    ion_concentration_ppm : float
        Target ion concentration in feed (mg/L = ppm)
    flow_rate_lpm : float
        Flow rate in liters per minute
    target_ion : str
        Ion to recover ('Li+', 'Nd3+', 'Co2+')
    pore_diameter : float
        Pore diameter in Angstroms (default: Z/2)

    Returns:
    --------
    dict : Revenue projection and system parameters
    """
    ion = IONS.get(target_ion, IONS['Li+'])

    # Calculate efficiency and selectivity
    efficiency = calculate_capture_efficiency(target_ion, pore_diameter)
    selectivity_Na = calculate_selectivity(target_ion, 'Na+', pore_diameter)

    # Volume calculations
    daily_volume_m3 = flow_rate_lpm * 60 * 24 / 1000
    annual_volume_m3 = daily_volume_m3 * 365

    # Mass recovery
    daily_recovery_kg = (ion_concentration_ppm * daily_volume_m3 *
                         efficiency * 1e-3)  # ppm × m³ × efficiency → kg
    annual_recovery_kg = daily_recovery_kg * 365

    # Revenue
    annual_revenue = annual_recovery_kg * ion['value_usd_kg']

    # For Li+, convert to Li2CO3 equivalent (market product)
    if target_ion == 'Li+':
        # Li2CO3 contains 18.8% Li by mass
        li2co3_kg = annual_recovery_kg / 0.188
        li2co3_price = 19  # $/kg Li2CO3 (2026)
        annual_revenue = li2co3_kg * li2co3_price

    return {
        'target_ion': target_ion,
        'pore_diameter_A': float(pore_diameter),
        'capture_efficiency': float(efficiency),
        'selectivity_vs_Na': float(selectivity_Na),
        'daily_volume_m3': float(daily_volume_m3),
        'annual_volume_m3': float(annual_volume_m3),
        'daily_recovery_kg': float(daily_recovery_kg),
        'annual_recovery_kg': float(annual_recovery_kg),
        'annual_revenue_USD': float(annual_revenue),
        'revenue_per_m3_USD': float(annual_revenue / annual_volume_m3)
    }

def optimize_pore_diameter(target_ion: str) -> float:
    """
    Find optimal pore diameter for target ion using Z-harmonics.

    Tests Z, Z/2, Z/3, 2Z to find best geometric match.
    """
    harmonics = {
        'Z': Z,
        'Z/2': Z/2,
        'Z/3': Z/3,
        '2Z': 2*Z
    }

    best_pore = Z/2
    best_efficiency = 0

    for name, diameter in harmonics.items():
        eff = calculate_capture_efficiency(target_ion, diameter)
        if eff > best_efficiency:
            best_efficiency = eff
            best_pore = diameter

    return best_pore

# Example usage
if __name__ == '__main__':
    print("="*60)
    print("Z-MINING REVENUE PROJECTION")
    print("="*60)

    # Standard brine input
    result = project_annual_revenue(
        ion_concentration_ppm=150,
        flow_rate_lpm=15,
        target_ion='Li+',
        pore_diameter=Z_HALF
    )

    print(f"\nTarget: {result['target_ion']}")
    print(f"Pore diameter: {result['pore_diameter_A']:.2f} Å (Z/2)")
    print(f"Capture efficiency: {result['capture_efficiency']:.1%}")
    print(f"Selectivity (Li/Na): {result['selectivity_vs_Na']:.0f}:1")
    print(f"\nDaily recovery: {result['daily_recovery_kg']:.1f} kg")
    print(f"Annual recovery: {result['annual_recovery_kg']:.0f} kg")
    print(f"\nAnnual revenue: ${result['annual_revenue_USD']:,.0f}")
    print(f"Revenue per m³: ${result['revenue_per_m3_USD']:.2f}")
```
